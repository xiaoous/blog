## 背景

本文基于**江协科技 C51 单片机教程**的「红外传感器篇」。那一篇讲的是 C51 上的原理，这里把它搬到 STM32 上：用定时器输入捕获测量红外信号的脉宽，按 NEC 协议把遥控器的地址码、命令码解出来。

代码部分给出了 HAL 库驱动的完整实现，可以直接移植；标准库版本见文末网盘。

> 想真正弄懂原理，建议先去看江协科技 C51 教程的红外传感器篇。

## 一、前期准备与接线

## 二、NEC 协议与时序

一次按键的完整帧由这几段组成（下面的脉宽为**下降沿到下降沿**的间隔，也就是代码里捕获到的值）：

| 段 | 间隔 | 含义 |
| --- | --- | --- |
| 引导码 | ≈ 13.5ms（9ms 低 + 4.5ms 高） | 一帧开始 |
| 数据位 0 | ≈ 1.12ms（560µs 低 + 560µs 高） | bit = 0 |
| 数据位 1 | ≈ 2.25ms（560µs 低 + 1.69ms 高） | bit = 1 |

一帧共 4 个字节，依次是：**地址码、地址反码、命令码、命令反码**。反码用来校验，只有 `address == ~address_inv` 且 `command == ~command_inv` 这一帧才算有效。

解码思路：

1. 定时器配成 1µs 计数精度（72MHz 时钟、72 分频）；
2. 每次捕获中断读 `CCR3`，得到与上一次捕获之间的间隔；
3. 判断间隔落在哪个区间：引导码 / 位 0 / 位 1 / 无效；
4. 每 8 位拼成一个字节，收满 4 字节就置位「接收完成」标志；
5. 主循环里取数据，校验通过再拿去用。

## 三、代码实现

### 3.1 IR1.c

```c
/* USER CODE BEGIN Header */
/**
  ******************************************************************************
  * @file    IR1.c
  * @brief   红外接收模块实现
  *          支持NEC协议红外遥控器解码
  ******************************************************************************
  * @attention
  *
  * Copyright (c) 2025 STMicroelectronics.
  * All rights reserved.
  *
  ******************************************************************************
  */
/* USER CODE END Header */

/* Includes ------------------------------------------------------------------*/
#include "IR1.h"
#include <string.h>  // 只包含string.h用于memset函数

/* USER CODE BEGIN 0 */

/* USER CODE END 0 */

// 红外模块私有变量
static uint16_t ir_time_us = 0;
static uint8_t ir_cnt = 0;
static uint8_t ir_temp = 0;
static uint8_t ir_buf[4] = {0};
static uint8_t ir_data_ready = 0;  // 红外数据接收完成标志
static IR_Data_t ir_data = {0};    // 红外数据结构

/* USER CODE BEGIN 1 */

/* USER CODE END 1 */

/* Private functions --------------------------------------------------------*/

/**
  * @brief  红外数据校验
  * @param  buf: 原始数据缓冲区
  * @retval uint8_t: 校验结果 (1:有效, 0:无效)
  */
static uint8_t IR1_ValidateData(uint8_t *buf)
{
    // 校验地址码和地址反码
    if (buf[0] != (uint8_t)~buf[1])
        return 0;

    // 校验命令码和命令反码
    if (buf[2] != (uint8_t)~buf[3])
        return 0;

    return 1;
}

/**
  * @brief  解析红外数据
  * @param  buf: 原始数据缓冲区
  * @param  data: 解析后的数据存储结构
  * @retval None
  */
static void IR1_ParseData(uint8_t *buf, IR_Data_t *data)
{
    data->address = buf[0];
    data->address_inv = buf[1];
    data->command = buf[2];
    data->command_inv = buf[3];
    data->timestamp = HAL_GetTick();  // 获取当前时间戳
    data->valid = 1;
}

/**
  * @brief  重置红外接收状态
  * @retval None
  */
static void IR1_ResetReceiver(void)
{
    ir_cnt = 0;
    ir_temp = 0;
    memset(ir_buf, 0, sizeof(ir_buf));
}

/* USER CODE BEGIN 2 */

/* USER CODE END 2 */

/* Public functions --------------------------------------------------------*/

/**
  * @brief  红外模块初始化
  * @retval None
  */
void IR1_Init(void)
{
    // 初始化变量
    ir_time_us = 0;
    ir_cnt = 0;
    ir_temp = 0;
    ir_data_ready = 0;

    // 清零数据结构
    memset(&ir_data, 0, sizeof(IR_Data_t));
    memset(ir_buf, 0, sizeof(ir_buf));

    // 注意：定时器初始化应在MX_TIM3_Init()中完成
    // 这里只是启动定时器输入捕获
    HAL_TIM_IC_Start_IT(&htim3, TIM_CHANNEL_3);
    __HAL_TIM_SET_CAPTUREPOLARITY(&htim3, TIM_CHANNEL_3, TIM_ICPOLARITY_FALLING);
    TIM3->CNT = 0;
}

/**
  * @brief  获取红外数据
  * @param  data: 红外数据存储结构指针
  * @retval uint8_t: 获取结果 (1:成功, 0:无数据)
  */
uint8_t IR1_GetData(IR_Data_t *data)
{
    if (!ir_data_ready || !data)
        return 0;

    // 验证数据有效性
    if (!IR1_ValidateData(ir_buf))
        return 0;

    // 解析数据
    IR1_ParseData(ir_buf, data);

    return 1;
}

/**
  * @brief  检查红外数据是否就绪
  * @retval uint8_t: 数据就绪状态 (1:就绪, 0:未就绪)
  */
uint8_t IR1_IsDataReady(void)
{
    return ir_data_ready;
}

/**
  * @brief  清除红外数据就绪标志
  * @retval None
  */
void IR1_ClearFlag(void)
{
    ir_data_ready = 0;
}

/**
  * @brief  红外解码回调函数
  * @param  htim: 定时器句柄
  * @retval None
  */
void HAL_TIM_IC_CaptureCallback(TIM_HandleTypeDef *htim)
{
    if (htim != &htim3)
        return;

    // 读取捕获值
    ir_time_us = TIM3->CCR3;

    // 识别引导码
    if (ir_time_us >= IR_LEADING_CODE_MIN && ir_time_us <= IR_LEADING_CODE_MAX)
    {
        IR1_ResetReceiver();
    }
    // 识别数据位0
    else if (ir_time_us >= IR_BIT0_MIN && ir_time_us <= IR_BIT0_MAX)
    {
        ir_cnt++;
        ir_temp >>= 1;
        ir_temp &= 0x7F;  // 确保最高位为0
    }
    // 识别数据位1
    else if (ir_time_us >= IR_BIT1_MIN && ir_time_us <= IR_BIT1_MAX)
    {
        ir_cnt++;
        ir_temp >>= 1;
        ir_temp |= 0x80;  // 设置最高位为1
    }
    // 无效信号，重置接收
    else
    {
        IR1_ResetReceiver();
    }

    // 存储接收到的字节
    if (ir_cnt == 8)
        ir_buf[0] = ir_temp;
    else if (ir_cnt == 16)
        ir_buf[1] = ir_temp;
    else if (ir_cnt == 24)
        ir_buf[2] = ir_temp;
    else if (ir_cnt == 32)
    {
        ir_buf[3] = ir_temp;
        ir_data_ready = 1;  // 设置数据接收完成标志
    }

    // 重置定时器计数器
    TIM3->CNT = 0;
}

/* USER CODE BEGIN 3 */

/* USER CODE END 3 */
```

### 3.2 IR1.h 里需要的定义

原篇没有贴头文件，这里按 NEC 的典型值补一份，实测时要根据自己遥控器微调窗口：

```c
#ifndef __IR1_H
#define __IR1_H

#include "main.h"
#include "tim.h"   // htim3 的声明所在，按自己工程的实际位置改

/* 时间窗口，单位 µs：按「下降沿到下降沿」的间隔判断 */
#define IR_LEADING_CODE_MIN   13000
#define IR_LEADING_CODE_MAX   14500
#define IR_BIT0_MIN           900
#define IR_BIT0_MAX           1400
#define IR_BIT1_MIN           1800
#define IR_BIT1_MAX           2600

typedef struct
{
    uint8_t  address;
    uint8_t  address_inv;
    uint8_t  command;
    uint8_t  command_inv;
    uint32_t timestamp;
    uint8_t  valid;
} IR_Data_t;

void    IR1_Init(void);
uint8_t IR1_GetData(IR_Data_t *data);
uint8_t IR1_IsDataReady(void);
void    IR1_ClearFlag(void);

#endif
```

## 四、使用说明

1. 下载后解压，把 `IR1.c` 和 `IR1.h` 添加到 STM32 工程里；
2. 配置 TIM3 为输入捕获模式：时钟 72MHz，预分频 72，计数精度 1µs；
3. 编译前确认 HAL 库环境配置正确（适配自己的 STM32 型号）；
4. 如果要改引脚或定时器通道，代码里的 `htim3`、`TIM_CHANNEL_3` 和 `CCR3` 要同步修改。

主循环里的用法大致是：

```c
IR_Data_t ir;

IR1_Init();          /* 初始化一次 */

while (1)
{
    if (IR1_GetData(&ir))          /* 取到一帧有效数据 */
    {
        IR1_ClearFlag();
        /* ir.command 就是按键码，按需处理 */
    }
}
```
